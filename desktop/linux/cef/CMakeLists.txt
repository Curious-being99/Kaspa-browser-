cmake_minimum_required(VERSION 3.21)
project(kaspabrowser_cef LANGUAGES C CXX)

set(CEF_ROOT "" CACHE PATH "Extracted CEF binary distribution root")
if(NOT CEF_ROOT OR NOT EXISTS "${CEF_ROOT}/include/cef_app.h")
  message(FATAL_ERROR "Set CEF_ROOT to an extracted Linux CEF binary distribution")
endif()
if(NOT EXISTS "${CEF_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "CEF_ROOT must contain the CEF distribution CMakeLists.txt")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${CEF_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/cef")

add_executable(kaspabrowser-cef src/main.cc src/browser_host.cc)
target_include_directories(kaspabrowser-cef PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(kaspabrowser-cef PRIVATE libcef_dll_wrapper libcef_lib)
if(TARGET cef_sandbox)
  target_link_libraries(kaspabrowser-cef PRIVATE cef_sandbox)
endif()

set_target_properties(kaspabrowser-cef PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(resource_dir Release Resources)
  if(EXISTS "${CEF_ROOT}/${resource_dir}")
    add_custom_command(TARGET kaspabrowser-cef POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/${resource_dir}" "$<TARGET_FILE_DIR:kaspabrowser-cef>")
  endif()
endforeach()
